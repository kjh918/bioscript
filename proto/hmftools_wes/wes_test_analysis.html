fix NT result file name
SampleListFile="/storage/home/kangsm/analysis/gcx.pipeline.use.hmftools.in.wes/01.preprocess/analysis.list"
    
for TASK in $(seq 2 $(cat "${SampleListFile}" | wc -l))
do
    
    RunInfo=$(sed -n -e "${TASK} p" ${SampleListFile})
    BatchID=$(echo ${RunInfo} | awk '{print $3}')
    TumorSeqID=$(echo ${RunInfo} | awk '{print $1}')

    echo -e -n "${TumorSeqID}...."

    DataDir=${BaseDir}/${BatchID}
    TumorVcfDir=${DataDir}/${TumorSeqID}/vcf
    MutectDir_NT=${TumorVcfDir}/mutect2_matched_normal
    SomaticseqDir=${TumorVcfDir}/somaticseq_matched_normal
    PurecnVcfPrepDir=${TumorVcfDir}/purecn_input_matched_normal
    PurecnDir=${DataDir}/${TumorSeqID}/cnv/purecn/matched_normal
    PrefixTag="NT"
    OutputPrefix_Mutect_VCF=${MutectDir_NT}/${TumorSeqID}.${PrefixTag}.mutect2
    SomaticSeqInput_MUTECT2=${OutputPrefix_Mutect_VCF}.filtered.norm.vcf.gz

    bcftools annotate -a ${PurecnDir}/${TumorSeqID}.${PrefixTag}.variants.purecn.vcf.gz \
        -c FMT/GT,FMT/DP,FMT/AD,INFO/PureCN.PS,INFO/PureCN.ML.SOMATIC,INFO/PureCN.ML.C,INFO/PureCN.CS,INFO/PureCN.ML.LOH,INFO/PureCN.GHOMOZYGOUS,INFO/PureCN.LR,INFO/PureCN.CF,INFO/PureCN.ML.M \
        -Oz -o ${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.somaticseq.consensus.purecn.vcf.gz \
        ${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.somaticseq.consensus.pass.vcf.gz
    
    tabix -p vcf ${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.somaticseq.consensus.purecn.vcf.gz

    echo '##INFO=<ID=MUTECT2_FILTER,Number=1,Type=String,Description="Mutect2 VCF FILTER">' > ${PurecnVcfPrepDir}/mutect2.filter.header
    bcftools query -f '%CHROM\t%POS\t%REF\t%ALT\t%FILTER\n' "${SomaticSeqInput_MUTECT2}" \
        | sort -k1,1V -k2,2n  | bgzip -c > "${PurecnVcfPrepDir}/mutect2.filter.tsv.gz"
    tabix -s 1 -b 2 -e 2 "${PurecnVcfPrepDir}/mutect2.filter.tsv.gz"

    mv "${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.somaticseq.consensus.filtered.vcf.gz" "${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.somaticseq.consensus.purecn.vcf.gz"
    mv "${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.somaticseq.consensus.filtered.vcf.gz.tbi" "${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.somaticseq.consensus.purecn.vcf.gz.tbi"

    bcftools annotate -a "${PurecnVcfPrepDir}/mutect2.filter.tsv.gz" \
        -h "${PurecnVcfPrepDir}/mutect2.filter.header" \
        -c CHROM,POS,REF,ALT,INFO/MUTECT2_FILTER \
        -Oz -o "${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.somaticseq.consensus.purecn.mutect2.filter.added.vcf.gz" \
        "${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.somaticseq.consensus.purecn.vcf.gz"
    tabix -p vcf "${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.somaticseq.consensus.purecn.mutect2.filter.added.vcf.gz"

    ln -Tsf "${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.somaticseq.consensus.purecn.mutect2.filter.added.vcf.gz" ${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.VARIANTS.vcf.gz
    ln -Tsf "${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.somaticseq.consensus.purecn.mutect2.filter.added.vcf.gz.tbi" ${TumorVcfDir}/${TumorSeqID}.${PrefixTag}.VARIANTS.vcf.gz.tbi

    echo "DONE"
done
